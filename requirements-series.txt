# Optional dependencies for the time-series foundation-model comparators.
#
# DELIBERATELY NOT IN requirements.txt. That file is installed by Render, by
# the daily pipeline and by the weekly evaluation, none of which call this
# code. torch was removed in Phase 0 as the largest contributor to memory
# pressure on the free-tier instance that had already been OOM-killed once;
# putting it back into the shared file would undo that for no benefit.
#
# Installed only by .github/workflows/series-comparison.yml, on top of
# requirements.txt. `pipeline/chronos_forecaster.py` imports torch lazily, so
# nothing on the API or daily path can pull it in by accident.
#
# Install locally with:
#   pip install -r requirements.txt -r requirements-series.txt

--extra-index-url https://download.pytorch.org/whl/cpu

# CPU-only. The GPU wheel is ~2.5GB against ~200MB and there is no GPU on a
# GitHub-hosted runner, so the default index would download CUDA for nothing.
torch>=2.2,<3

# chronos-forecasting pins transformers itself (>=4.41,<6); it also pulls
# accelerate and einops.
chronos-forecasting>=2.0

# TimesFM-2.5 ships INSIDE transformers as TimesFm2_5ModelForPrediction, so it
# costs no package of its own — but the class landed in 5.x, and chronos' pin
# alone would happily resolve 4.x and leave the symbol missing. Pinned here so
# the failure is at install time rather than an ImportError an hour into a run.
#
# The `timesfm` PyPI package is deliberately NOT used: it is still at 2.0.2, so
# reaching 2.5 through it needs a git clone plus `pip install -e`, which is an
# unpinned dependency in CI.
transformers>=5.0,<6
